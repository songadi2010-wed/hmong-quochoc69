        function projectOrbitPoint(angle, radius, tiltX, tiltZ, cx, cy) {
            let x = Math.cos(angle) * radius;
            let y = 0;
            let z = Math.sin(angle) * radius;

            let y1 = y * Math.cos(tiltX) - z * Math.sin(tiltX);
            let z1 = y * Math.sin(tiltX) + z * Math.cos(tiltX);

            let x2 = x * Math.cos(tiltZ) - y1 * Math.sin(tiltZ);
            let y2 = x * Math.sin(tiltZ) + y1 * Math.cos(tiltZ);

            const perspective = 650 / (650 + z1);
            return {
                x: cx + x2 * perspective,
                y: cy + y2 * perspective,
                z: z1,
                scale: perspective
            };
        }

        function drawSatelliteSystem(time) {
            satCtx.clearRect(0, 0, width, height);

            const cx = width * 0.5;
            const cy = height * 0.54;
            const baseRadius = Math.min(width, height) * 0.18;

            satCtx.save();
            satCtx.translate(cx, cy);
            satCtx.strokeStyle = 'rgba(0, 255, 65, 0.14)';
            satCtx.lineWidth = 1;
            satCtx.setLineDash([4, 8]);
            satCtx.beginPath();
            satCtx.arc(0, 0, baseRadius * 2.05, 0, Math.PI * 2);
            satCtx.stroke();
            satCtx.setLineDash([]);

            for (let i = 0; i < 36; i++) {
                const a = (i * Math.PI) / 18 + globeRotation * 0.2;
                const r1 = baseRadius * 2.02;
                const r2 = baseRadius * (i % 3 === 0 ? 2.12 : 2.07);
                satCtx.beginPath();
                satCtx.moveTo(Math.cos(a) * r1, Math.sin(a) * r1);
                satCtx.lineTo(Math.cos(a) * r2, Math.sin(a) * r2);
                satCtx.strokeStyle = i % 3 === 0 ? 'rgba(0, 255, 65, 0.35)' : 'rgba(0, 255, 65, 0.15)';
                satCtx.stroke();
            }
            satCtx.restore();

            satellites.forEach(sat => {
                sat.angle += sat.speed;
                const orbitR = baseRadius * sat.radiusMult;

                satCtx.beginPath();
                for (let step = 0; step <= 64; step++) {
                    const a = (step / 64) * Math.PI * 2;
                    const pt = projectOrbitPoint(a, orbitR, sat.tiltX, sat.tiltZ, cx, cy);
                    if (step === 0) satCtx.moveTo(pt.x, pt.y);
                    else satCtx.lineTo(pt.x, pt.y);
                }
                satCtx.strokeStyle = 'rgba(0, 255, 65, 0.16)';
                satCtx.lineWidth = 1;
                satCtx.stroke();

                const pos = projectOrbitPoint(sat.angle, orbitR, sat.tiltX, sat.tiltZ, cx, cy);
                sat.currentPos = pos;

                sat.trail.push({ x: pos.x, y: pos.y, z: pos.z });
                if (sat.trail.length > 26) sat.trail.shift();

                if (pos.z > 0) {
                    drawSatelliteTrail(sat);
                    drawDetailedSatellite(satCtx, pos.x, pos.y, sat.angle, pos.scale, sat);
                }
            });

            globeRotation += 0.005;
            satCtx.save();
            satCtx.translate(cx, cy);

            const earthGlow = satCtx.createRadialGradient(0, 0, baseRadius * 0.2, 0, 0, baseRadius * 1.25);
            earthGlow.addColorStop(0, 'rgba(0, 255, 65, 0.22)');
            earthGlow.addColorStop(0.7, 'rgba(0, 180, 50, 0.08)');
            earthGlow.addColorStop(1, 'rgba(0, 255, 65, 0)');
            satCtx.fillStyle = earthGlow;
            satCtx.beginPath();
            satCtx.arc(0, 0, baseRadius * 1.25, 0, Math.PI * 2);
            satCtx.fill();

            satCtx.strokeStyle = 'rgba(0, 255, 65, 0.45)';
            satCtx.lineWidth = 1.5;
            satCtx.beginPath();
            satCtx.arc(0, 0, baseRadius, 0, Math.PI * 2);
            satCtx.stroke();

            const meridians = 8;
            for (let i = 0; i < meridians; i++) {
                const lon = (i / meridians) * Math.PI + globeRotation;
                const rx = Math.abs(Math.cos(lon)) * baseRadius;
                satCtx.beginPath();
                satCtx.ellipse(0, 0, rx, baseRadius, 0, 0, Math.PI * 2);
                satCtx.strokeStyle = 'rgba(0, 255, 65, 0.18)';
                satCtx.lineWidth = 1;
                satCtx.stroke();
            }

            const parallels = [-0.6, -0.3, 0, 0.3, 0.6];
            parallels.forEach(lat => {
                const y = lat * baseRadius;
                const r = Math.sqrt(Math.max(0, baseRadius * baseRadius - y * y));
                satCtx.beginPath();
                satCtx.ellipse(0, y, r, r * 0.22, 0, 0, Math.PI * 2);
                satCtx.strokeStyle = lat === 0 ? 'rgba(0, 255, 65, 0.42)' : 'rgba(0, 255, 65, 0.16)';
                satCtx.stroke();
            });

            const beaconAngle = globeRotation * 1.5;
            const bx = Math.cos(beaconAngle) * baseRadius * 0.55;
            const by = -baseRadius * 0.22;
            if (Math.sin(beaconAngle) > -0.2) {
                const pulse = (Math.sin(time * 0.006) + 1) * 0.5;
                satCtx.fillStyle = '#a8ffbe';
                satCtx.beginPath();
                satCtx.arc(bx, by, 3, 0, Math.PI * 2);
                satCtx.fill();

                satCtx.strokeStyle = `rgba(0, 255, 65, ${0.8 - pulse * 0.7})`;
                satCtx.lineWidth = 1.5;
                satCtx.beginPath();
                satCtx.arc(bx, by, 4 + pulse * 14, 0, Math.PI * 2);
                satCtx.stroke();
            }

            satCtx.restore();

            satellites.forEach(sat => {
                const pos = sat.currentPos;
                if (pos && pos.z <= 0) {
                    drawSatelliteTrail(sat);

                    const beamGrad = satCtx.createLinearGradient(pos.x, pos.y, cx, cy);
                    beamGrad.addColorStop(0, 'rgba(0, 255, 65, 0.45)');
                    beamGrad.addColorStop(1, 'rgba(0, 255, 65, 0.02)');
                    satCtx.strokeStyle = beamGrad;
                    satCtx.lineWidth = 1;
                    satCtx.setLineDash([3, 5]);
                    satCtx.beginPath();
                    satCtx.moveTo(pos.x, pos.y);
                    satCtx.lineTo(cx, cy);
                    satCtx.stroke();
                    satCtx.setLineDash([]);

                    drawDetailedSatellite(satCtx, pos.x, pos.y, sat.angle, pos.scale, sat);
                }
            });
        }

        function drawSatelliteTrail(sat) {
            if (sat.trail.length < 2) return;
            for (let i = 1; i < sat.trail.length; i++) {
                const p0 = sat.trail[i - 1];
                const p1 = sat.trail[i];
                const alpha = (i / sat.trail.length) * 0.55;
                satCtx.strokeStyle = `rgba(0, 255, 65, ${alpha})`;
                satCtx.lineWidth = (i / sat.trail.length) * 2.2;
                satCtx.beginPath();
                satCtx.moveTo(p0.x, p0.y);
                satCtx.lineTo(p1.x, p1.y);
                satCtx.stroke();
            }
        }

        class RealisticSpider {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.angle = -Math.PI / 4;
                this.targetX = x;
                this.targetY = y;
                this.hasUserTarget = false;
                this.idleTimer = 0;

                this.scale = Math.min(width, height) < 600 ? 0.85 : 1.05;
                this.speed = 0;
                this.maxSpeed = 3.6;

                this.legs = [];
                const legConfigs = [
                    { side: -1, baseAngle: -0.52, reach: 78, l1: 36, l2: 42, l3: 32, group: 0 },
                    { side:  1, baseAngle:  0.52, reach: 78, l1: 36, l2: 42, l3: 32, group: 1 },
                    { side: -1, baseAngle: -1.18, reach: 74, l1: 34, l2: 38, l3: 30, group: 1 },
                    { side:  1, baseAngle:  1.18, reach: 74, l1: 34, l2: 38, l3: 30, group: 0 },
                    { side: -1, baseAngle: -1.85, reach: 72, l1: 33, l2: 37, l3: 29, group: 0 },
                    { side:  1, baseAngle:  1.85, reach: 72, l1: 33, l2: 37, l3: 29, group: 1 },
                    { side: -1, baseAngle: -2.52, reach: 84, l1: 38, l2: 44, l3: 34, group: 1 },
                    { side:  1, baseAngle:  2.52, reach: 84, l1: 38, l2: 44, l3: 34, group: 0 }
                ];

                legConfigs.forEach((cfg, idx) => {
                    const ideal = this.getIdealFootPos(cfg);
                    this.legs.push({
                        ...cfg,
                        index: idx,
                        footX: ideal.x,
                        footY: ideal.y,
                        stepStartX: ideal.x,
                        stepStartY: ideal.y,
                        stepTargetX: ideal.x,
                        stepTargetY: ideal.y,
                        stepping: false,
                        stepProgress: 0,
                        liftHeight: 0
                    });
                });
            }

            setTarget(tx, ty) {
                this.targetX = tx;
                this.targetY = ty;
                this.hasUserTarget = true;
                if (targetCoordsEl) {
                    targetCoordsEl.textContent = `X:${Math.round(tx)} Y:${Math.round(ty)}`;
                }
                if (spiderStatusEl) {
                    spiderStatusEl.textContent = `ĐANG DI CHUYỂN TỚI [${Math.round(tx)}, ${Math.round(ty)}]`;
                }
            }

            getIdealFootPos(cfg) {
                const worldAngle = this.angle + cfg.baseAngle;
                const dist = cfg.reach * this.scale;
                return {
                    x: this.x + Math.cos(worldAngle) * dist,
                    y: this.y + Math.sin(worldAngle) * dist
                };
            }

            update(time) {
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const dist = Math.hypot(dx, dy);

                if (!this.hasUserTarget && dist < 12) {
                    this.idleTimer++;
                    if (this.idleTimer > 180) {
                        this.idleTimer = 0;
                        const margin = 110;
                        this.targetX = margin + Math.random() * (width - margin * 2);
                        this.targetY = Math.max(150, margin + Math.random() * (height - margin * 2));
                    }
                }

                if (dist > 6) {
                    const desiredAngle = Math.atan2(dy, dx);
                    let angleDiff = desiredAngle - this.angle;
                    while (angleDiff > Math.PI) angleDiff -= Math.PI * 2;
                    while (angleDiff < -Math.PI) angleDiff += Math.PI * 2;

                    this.angle += angleDiff * 0.11;

                    const turnFactor = Math.max(0.25, 1 - Math.abs(angleDiff) / Math.PI);
                    const arriveFactor = Math.min(1, dist / 65);
                    const targetSpeed = this.maxSpeed * turnFactor * arriveFactor;
                    this.speed += (targetSpeed - this.speed) * 0.15;

                    this.x += Math.cos(this.angle) * this.speed;
                    this.y += Math.sin(this.angle) * this.speed;
                } else {
                    this.speed *= 0.8;
                    if (this.hasUserTarget && this.speed < 0.15) {
                        this.hasUserTarget = false;
                        this.idleTimer = 0;
                        if (spiderStatusEl) {
                            spiderStatusEl.textContent = "ĐÃ ĐẾN MỤC TIÊU // CHẾ ĐỘ CẢNH GIỚI";
                        }
                    }
                }

                const anySteppingInGroup = [
                    this.legs.some(l => l.group === 0 && l.stepping),
                    this.legs.some(l => l.group === 1 && l.stepping)
                ];

                this.legs.forEach(leg => {
                    const ideal = this.getIdealFootPos(leg);
                    const leadX = ideal.x + Math.cos(this.angle) * (this.speed * 6.5);
                    const leadY = ideal.y + Math.sin(this.angle) * (this.speed * 6.5);

                    const errDist = Math.hypot(leadX - leg.footX, leadY - leg.footY);
                    const stepThreshold = (24 + (leg.index % 2) * 5) * this.scale;

                    const otherGroup = leg.group === 0 ? 1 : 0;
                    if (!leg.stepping && errDist > stepThreshold && !anySteppingInGroup[otherGroup]) {
                        leg.stepping = true;
                        leg.stepProgress = 0;
                        leg.stepStartX = leg.footX;
                        leg.stepStartY = leg.footY;
                        leg.stepTargetX = leadX;
                        leg.stepTargetY = leadY;
                    }

                    if (leg.stepping) {
                        const stepInc = Math.max(0.11, Math.min(0.24, 0.11 + this.speed * 0.032));
                        leg.stepProgress += stepInc;

                        if (leg.stepProgress >= 1) {
                            leg.stepProgress = 1;
                            leg.stepping = false;
                            leg.footX = leg.stepTargetX;
                            leg.footY = leg.stepTargetY;
                            leg.liftHeight = 0;
                        } else {
                            const t = leg.stepProgress;
                            const ease = t * t * (3 - 2 * t);
                            leg.footX = leg.stepStartX + (leg.stepTargetX - leg.stepStartX) * ease;
                            leg.footY = leg.stepStartY + (leg.stepTargetY - leg.stepStartY) * ease;
                            leg.liftHeight = Math.sin(t * Math.PI) * 14 * this.scale;
                        }
                    }
                });
            }

            drawRealisticLeg(ctx, leg, bodyBob) {
                const s = this.scale;

                const hipAngle = this.angle + leg.baseAngle * 0.65;
                const hipX = this.x + Math.cos(hipAngle) * (9 * s);
                const hipY = (this.y - bodyBob) + Math.sin(hipAngle) * (8 * s);

                const footX = leg.footX;
                const footY = leg.footY - leg.liftHeight;

                ctx.strokeStyle = 'rgba(0, 0, 0, 0.45)';
                ctx.lineWidth = 2.5 * s;
                ctx.lineCap = 'round';
                ctx.beginPath();
                ctx.moveTo(hipX + 4 * s, hipY + 7 * s);
                ctx.lineTo(leg.footX + 4 * s, leg.footY + 7 * s);
                ctx.stroke();

                const vx = footX - hipX;
                const vy = footY - hipY;
                const len = Math.hypot(vx, vy) || 1;
                const nx = (-vy / len) * leg.side;
                const ny = (vx / len) * leg.side;

                const kneeArch = (22 * s) + leg.liftHeight * 0.7;
                const kneeX = hipX + vx * 0.42 + nx * (10 * s);
                const kneeY = hipY + vy * 0.42 + ny * (10 * s) - kneeArch;

                const ankleX = hipX + vx * 0.76 + nx * (5 * s);
                const ankleY = hipY + vy * 0.76 + ny * (5 * s) - kneeArch * 0.42;

                ctx.strokeStyle = '#0c2413';
                ctx.lineWidth = 4.2 * s;
                ctx.lineCap = 'round';
                ctx.beginPath();
                ctx.moveTo(hipX, hipY);
                ctx.lineTo(kneeX, kneeY);
                ctx.stroke();

                ctx.strokeStyle = 'rgba(0, 255, 65, 0.55)';
                ctx.lineWidth = 1.2 * s;
                ctx.stroke();

                ctx.strokeStyle = '#11301a';
                ctx.lineWidth = 3.0 * s;
                ctx.beginPath();
                ctx.moveTo(kneeX, kneeY);
                ctx.lineTo(ankleX, ankleY);
                ctx.stroke();

                ctx.strokeStyle = 'rgba(0, 255, 65, 0.45)';
                ctx.lineWidth = 1.0 * s;
                ctx.stroke();

                ctx.strokeStyle = '#184726';
                ctx.lineWidth = 1.8 * s;
                ctx.beginPath();
                ctx.moveTo(ankleX, ankleY);
                ctx.lineTo(footX, footY);
                ctx.stroke();

                ctx.fillStyle = '#00ff41';
                ctx.beginPath();
                ctx.arc(kneeX, kneeY, 1.6 * s, 0, Math.PI * 2);
                ctx.arc(ankleX, ankleY, 1.2 * s, 0, Math.PI * 2);
                ctx.fill();

                if (!leg.stepping) {
                    ctx.fillStyle = 'rgba(0, 255, 65, 0.75)';
                    ctx.beginPath();
                    ctx.arc(footX, footY, 1.5 * s, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
        }

        resizeAllCanvases();
        const spider = new RealisticSpider(width * 0.5, height * 0.62);

        // ============================================================================
        // PHẦN 5: TƯƠNG TÁC CHẠM / CLICK MÀN HÌNH & CHẶN LAN SỰ KIỆN TỪ NÚT ZALO
        // ============================================================================
        function handleScreenInteract(clientX, clientY) {
            spider.setTarget(clientX, clientY);

            clickRipples.push({
                x: clientX,
                y: clientY,
                radius: 4,
                maxRadius: 48,
                alpha: 1.0
            });
        }

        spiderCanvas.addEventListener('pointerdown', (e) => {
            if (e.target.closest && e.target.closest('#zaloContactBtn')) return;
            e.preventDefault();
            handleScreenInteract(e.clientX, e.clientY);
        });

        const zaloBtn = document.getElementById('zaloContactBtn');
        if (zaloBtn) {
            ['pointerdown', 'mousedown', 'touchstart', 'click'].forEach(evtName => {
                zaloBtn.addEventListener(evtName, (e) => {
                    e.stopPropagation();
                });
            });
            zaloBtn.addEventListener('click', () => {
                if (spiderStatusEl) {
                    spiderStatusEl.textContent = "ĐANG MỞ ZALO LIÊN HỆ: 0378257342...";
                }
            });
        }

        function drawClickRipples(ctx) {
            for (let i = clickRipples.length - 1; i >= 0; i--) {
                const r = clickRipples[i];
                r.radius += 1.6;
                r.alpha -= 0.022;

                if (r.alpha <= 0) {
                    clickRipples.splice(i, 1);
                    continue;
                }

                ctx.save();
                ctx.translate(r.x, r.y);

                ctx.strokeStyle = `rgba(0, 255, 65, ${r.alpha})`;
                ctx.lineWidth = 1.6;
                ctx.beginPath();
                ctx.arc(0, 0, r.radius, 0, Math.PI * 2);
                ctx.stroke();

                ctx.strokeStyle = `rgba(168, 255, 190, ${r.alpha * 0.8})`;
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.arc(0, 0, r.radius * 0.45, 0, Math.PI * 2);
                ctx.stroke();

                const ch = 10;
                ctx.beginPath();
                ctx.moveTo(-ch, 0); ctx.lineTo(ch, 0);
                ctx.moveTo(0, -ch); ctx.lineTo(0, ch);
                ctx.stroke();

                ctx.restore();
            }

            if (spider.hasUserTarget) {
                ctx.save();
                ctx.translate(spider.targetX, spider.targetY);
                ctx.strokeStyle = 'rgba(0, 255, 65, 0.65)';
                ctx.setLineDash([3, 3]);
                ctx.beginPath();
                ctx.arc(0, 0, 14, 0, Math.PI * 2);
                ctx.stroke();
                ctx.setLineDash([]);
                ctx.restore();

                ctx.save();
                ctx.strokeStyle = 'rgba(0, 255, 65, 0.22)';
                ctx.lineWidth = 1;
                ctx.setLineDash([4, 6]);
                ctx.beginPath();
                ctx.moveTo(spider.x, spider.y);
                ctx.lineTo(spider.targetX, spider.targetY);
                ctx.stroke();
                ctx.restore();
            }
        }

        // ============================================================================
        // VÒNG LẶP CHÍNH
        // ============================================================================
        let frameCount = 0;
        function mainLoop(timestamp) {
            frameCount++;

            if (frameCount % 2 === 0) {
                drawMatrixRain();
            }

            drawSatelliteSystem(timestamp || 0);

            spiderCtx.clearRect(0, 0, width, height);
            drawClickRipples(spiderCtx);
            spider.update(timestamp || 0);
            spider.draw(spiderCtx, timestamp || 0);

            requestAnimationFrame(mainLoop);
        }

        window.onload = function () {
            for (let i = 0; i < 18; i++) {
                appendContinuousCode();
            }
            requestAnimationFrame(mainLoop);
        };
